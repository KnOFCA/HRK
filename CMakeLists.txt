cmake_minimum_required(VERSION 3.20)
project(HRK VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(HRK_HARMONY "Build the native Harmony backend" OFF)
if(HRK_HARMONY)
  # The SDK supplies a Clang-only toolchain; its legacy --gcc-toolchain flag is unused.
  unset(CMAKE_CXX_COMPILER_EXTERNAL_TOOLCHAIN)
endif()
option(HRK_SANITIZE "Enable address/undefined sanitizers" OFF)
if(MSVC)
  add_compile_options(/W4 /WX /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  if(HRK_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
function(hrk_library name source)
  add_library(${name} STATIC ${source})
  target_include_directories(${name} PUBLIC ${PROJECT_SOURCE_DIR}/src)
  if(ARGN)
    target_link_libraries(${name} PUBLIC ${ARGN})
  endif()
endfunction()
hrk_library(hrk_base src/kernel/base/base.cpp)
foreach(api game platform render)
  add_library(hrk_${api}_api INTERFACE)
  target_link_libraries(hrk_${api}_api INTERFACE hrk_base)
endforeach()

target_link_libraries(hrk_platform_api INTERFACE hrk_render_api)

hrk_library(hrk_replay src/kernel/replay/replay.cpp hrk_base)
